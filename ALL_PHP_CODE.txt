EcoLoop - Pickups, vehicles and drivers (PHP + MySQL, no framework)

INSTALL (XAMPP / WAMP / MAMP)
1. Copy this folder into htdocs (XAMPP) or www (WAMP). Start Apache and MySQL.
2. phpMyAdmin > Import > database.sql   (creates the "ecoloop" database; it resets the tables)
3. Check config.php (MySQL host, user, password).
4. Create the demo accounts ONCE, then delete the script:
      browser : http://localhost/ecoloop_php/install_users.php
      or CLI  : php install_users.php
5. Open http://localhost/ecoloop_php/ and log in.

THE DRIVER USER
- An admin creates a driver in Admin > Users > "Create a driver account" (name, email, password).
  This adds the person to the collectors list AND creates the login linked to it.
- The admin assigns pickups + a vehicle in Admin > Collections.
- The driver logs in and lands on "My route" (driver/index.php):
    * sees only his own pickups: Today / Overdue / Upcoming, with address, note, materials, vehicle
    * "Start: I'm on the way" (Scheduled -> On the way): needs a vehicle and the pickup day to be today or earlier
    * "Mark as collected" (On the way -> Completed)
- A driver with active pickups cannot be deactivated until they are reassigned.

FILES
config.php, database.sql, install_users.php

admin/vehicles.php               admin: fleet
admin/users.php                  admin: create drivers, activate/deactivate, reset passwords

assets/style.css

RULES
- Roles: citizen, driver, admin. Each page checks the role (require_login).
- A vehicle in maintenance cannot be assigned; one vehicle cannot have two active pickups at the same date and slot.
- Prepared statements (PDO), HTML escaping (e()), CSRF token on every POST.
- Before going online: change all passwords, delete install_users.php, use HTTPS.

-- EcoLoop : pickups et véhicules (MySQL / MariaDB)
CREATE DATABASE IF NOT EXISTS ecoloop CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE ecoloop;

DROP TABLE IF EXISTS users;
DROP TABLE IF EXISTS collection_categories;
DROP TABLE IF EXISTS collections;
DROP TABLE IF EXISTS vehicles;
DROP TABLE IF EXISTS collectors;

CREATE TABLE collectors (
  id     INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  name   VARCHAR(100) NOT NULL UNIQUE,
  active TINYINT(1) NOT NULL DEFAULT 1
) ENGINE=InnoDB;

CREATE TABLE vehicles (
  id          INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  code        VARCHAR(20) NOT NULL UNIQUE,            -- ex. T-14, V-02
  type        ENUM('Truck','Van') NOT NULL DEFAULT 'Truck',
  capacity_kg INT UNSIGNED NOT NULL DEFAULT 1000,
  status      ENUM('Available','Maintenance') NOT NULL DEFAULT 'Available'
) ENGINE=InnoDB;

CREATE TABLE collections (
  id           INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  code         VARCHAR(20) NULL UNIQUE,                -- ex. C-1049
  owner_name   VARCHAR(100) NOT NULL,
  pickup_date  DATE NOT NULL,
  slot         VARCHAR(20) NOT NULL,
  address      VARCHAR(255) NOT NULL,
  note         TEXT NULL,
  status       ENUM('Scheduled','On the way','Completed','Cancelled') NOT NULL DEFAULT 'Scheduled',
  collector_id INT UNSIGNED NULL,
  vehicle_id   INT UNSIGNED NULL,
  created_at   TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_col_collector FOREIGN KEY (collector_id) REFERENCES collectors(id) ON DELETE SET NULL,
  CONSTRAINT fk_col_vehicle   FOREIGN KEY (vehicle_id)   REFERENCES vehicles(id)   ON DELETE SET NULL,
  INDEX idx_vehicle_slot (vehicle_id, pickup_date, slot),
  INDEX idx_owner (owner_name)
) ENGINE=InnoDB;

CREATE TABLE collection_categories (
  collection_id INT UNSIGNED NOT NULL,
  category      ENUM('plastic','paper','glass','metal','organic','electronics') NOT NULL,
  PRIMARY KEY (collection_id, category),
  CONSTRAINT fk_cc_collection FOREIGN KEY (collection_id) REFERENCES collections(id) ON DELETE CASCADE
) ENGINE=InnoDB;

-- Comptes de connexion (admin, chauffeur, citoyen). Un chauffeur est lié à un collecteur.
CREATE TABLE users (
  id            INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  name          VARCHAR(100) NOT NULL,
  email         VARCHAR(150) NOT NULL UNIQUE,
  password_hash VARCHAR(255) NOT NULL,
  role          ENUM('admin','driver','citizen') NOT NULL,
  collector_id  INT UNSIGNED NULL UNIQUE,              -- renseigné pour les chauffeurs
  active        TINYINT(1) NOT NULL DEFAULT 1,
  created_at    TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_user_collector FOREIGN KEY (collector_id) REFERENCES collectors(id) ON DELETE SET NULL
) ENGINE=InnoDB;

function db(): PDO
{
    static $pdo = null;
    if ($pdo === null) {
        $dsn = 'mysql:host=' . DB_HOST . ';dbname=' . DB_NAME . ';charset=utf8mb4';
        $pdo = new PDO($dsn, DB_USER, DB_PASS, [
            PDO::ATTR_ERRMODE            => PDO::ERRMODE_EXCEPTION,
            PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
            PDO::ATTR_EMULATE_PREPARES   => false,
        ]);
    }
    return $pdo;

function e($v): string
{
    return htmlspecialchars((string)$v, ENT_QUOTES, 'UTF-8');
}

/* ---------- CSRF ---------- */
function csrf_token(): string
{
    if (empty($_SESSION['csrf'])) {
        $_SESSION['csrf'] = bin2hex(random_bytes(32));
    }
    return $_SESSION['csrf'];
}
function csrf_field(): string
{
    return '<input type="hidden" name="csrf" value="' . e(csrf_token()) . '">';
}
function csrf_check(): void
{
    $ok = isset($_POST['csrf'], $_SESSION['csrf']) && hash_equals($_SESSION['csrf'], $_POST['csrf']);
    if (!$ok) {
        http_response_code(400);
        exit('Invalid CSRF token.');
    }
}

/* ---------- Messages flash ---------- */
function flash(string $msg, string $kind = 'ok'): void
{
    $_SESSION['flash'][] = ['msg' => $msg, 'kind' => $kind];
}
function flashes(): array
{
    $f = $_SESSION['flash'] ?? [];
    unset($_SESSION['flash']);
    return $f;
}
function redirect(string $url): void
{
    header('Location: ' . $url);
    exit;
}

/* ---------- Affichage ---------- */
function status_class(string $s): string
{
    return ['Scheduled' => 'info', 'On the way' => 'warn', 'Completed' => 'ok', 'Cancelled' => 'bad'][$s] ?? 'mute';
}
function status_badge(string $s): string
{
    return '<span class="badge ' . status_class($s) . '">' . e($s) . '</span>';
}
function cats_badges(?string $csv): string
{
    $out = '';
    foreach (array_filter(explode(',', (string)$csv)) as $k) {
        $out .= '<span class="cat c-' . e($k) . '">' . e(CATEGORIES[$k] ?? $k) . '</span> ';
    }
    return $out;
}
function fmt_date(string $d): string
{
    $ts = strtotime($d);
    $label = date('D j M', $ts);
    return $d === date('Y-m-d') ? 'Today · ' . $label : $label;
}
function vehicle_label(?string $code, ?string $type): string
{
    return $code ? ($type ?: 'Vehicle') . ' ' . $code : '—';
}

/* ---------- Requêtes ---------- */
const COLLECTION_SELECT = "
    SELECT c.*, co.name AS collector_name, v.code AS vehicle_code, v.type AS vehicle_type,
           GROUP_CONCAT(cc.category ORDER BY cc.category SEPARATOR ',') AS cats
    FROM collections c
    LEFT JOIN collectors co ON co.id = c.collector_id
    LEFT JOIN vehicles v ON v.id = c.vehicle_id
    LEFT JOIN collection_categories cc ON cc.collection_id = c.id";
const COLLECTION_GROUP = " GROUP BY c.id, co.name, v.code, v.type ";

/** Un véhicule est "occupé" s'il a déjà une collecte active à la même date et au même créneau. */
function vehicle_busy(int $vehicleId, string $date, string $slot, int $excludeId = 0): bool
{
    $st = db()->prepare(
        "SELECT COUNT(*) FROM collections
         WHERE vehicle_id = ? AND pickup_date = ? AND slot = ?
           AND status IN ('Scheduled','On the way') AND id <> ?"
    );
    $st->execute([$vehicleId, $date, $slot, $excludeId]);
    return (int)$st->fetchColumn() > 0;
}

<?php
require_once __DIR__ . '/helpers.php';

/** Utilisateur connecté (ou null). Rechargé depuis la base à chaque requête. */
function current_user(): ?array
{
    static $cache = false;
    if ($cache !== false) {
        return $cache;
    }
    if (empty($_SESSION['uid'])) {
        return $cache = null;
    }
    $st = db()->prepare("SELECT id, name, email, role, collector_id, active FROM users WHERE id = ?");
    $st->execute([$_SESSION['uid']]);
    $u = $st->fetch();
    if (!$u || !$u['active']) {
        unset($_SESSION['uid']);
        return $cache = null;
    }
    return $cache = $u;
}

/** Page d'accueil selon le rôle (chemin relatif à la racine du projet). */
function home_for(string $role): string
{
    return ['admin' => 'admin/collections.php', 'driver' => 'driver/index.php', 'citizen' => 'index.php'][$role] ?? 'index.php';
}

/** Exige une connexion et un des rôles donnés. $root = '' ou '../' selon le dossier de la page. */
function require_login(array $roles, string $root = ''): array
{
    $u = current_user();
    if (!$u) {
        redirect($root . 'login.php');
    }
    if (!in_array($u['role'], $roles, true)) {
        flash('Access denied for your account.', 'bad');
        redirect($root . home_for($u['role']));
    }
    return $u;
}

<?php
require_once __DIR__ . '/includes/auth.php';

if ($u = current_user()) {
    redirect(home_for($u['role']));
}

$error = null;
$email = '';
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    csrf_check();
    $email = strtolower(trim($_POST['email'] ?? ''));
    $pass  = (string)($_POST['password'] ?? '');
    $now   = time();

    // Anti brute-force simple : 5 échecs max par 5 minutes et par session
    $fails = array_values(array_filter($_SESSION['login_fail'] ?? [], fn($t) => $t > $now - 300));
    $_SESSION['login_fail'] = $fails;

    if (count($fails) >= 5) {
        $error = 'Too many attempts. Try again in a few minutes.';
    } else {
        $st = db()->prepare("SELECT id, password_hash, role, active FROM users WHERE email = ?");
        $st->execute([$email]);
        $row = $st->fetch();
        if ($row && $row['active'] && password_verify($pass, $row['password_hash'])) {
            session_regenerate_id(true);
            $_SESSION['uid'] = (int)$row['id'];
            unset($_SESSION['login_fail']);
            redirect(home_for($row['role']));
        }
        $_SESSION['login_fail'][] = $now;
        $error = 'Invalid email or password.';
    }
}

$title = 'Log in'; $active = ''; $root = '';
include __DIR__ . '/includes/header.php';
?>
<div class="login-box card">
  <h1>Log in</h1>
  <p class="muted">Citizens, drivers and admins use the same login.</p>
  <?php if ($error): ?><div class="flash bad"><?= e($error) ?></div><?php endif; ?>
  <form method="post" novalidate>
    <?= csrf_field() ?>

                }

        }
    } elseif ($action === 'reset_password') {
        $pass = (string)($_POST['password'] ?? '');
        if (mb_strlen($pass) < 8) {
            flash('New password must have at least 8 characters.', 'bad');

            $pdo->prepare("UPDATE users SET password_hash = ? WHERE id = ?")->execute([password_hash($pass, PASSWORD_DEFAULT), $id]);
            flash('Password updated.');
        }
    }
    redirect('users.php');
}
